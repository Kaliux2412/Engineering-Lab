
// SITUACIÓN PROBLEMA - ENTREGA FINAL
// Filtro de media móvil de 8 datos
// Calcula el promedio de ventanas consecutivas de 8 valores
// y almacena cada resultado en el arreglo Moving

Inicio,
    // Verifica si todavía quedan datos por procesar
    Load num_datos
    Subt ONE
    SkipCond 400
    Jump Inicializar
    Jump Fin

Inicializar,
    // Calcula la posición inicial de la ventana actual
    Load ptr_inicial
    Add inicio_movil
    Store ptr_Datos

    // Reinicia la suma acumulada
    Load suma
    Clear
    Store suma

    // Reinicia el contador utilizado para calcular el promedio
    Load resta
    Clear
    Store resta

    // Carga el primer dato de la ventana
    LoadI ptr_Datos
    Store suma
    Jump Contador

Contador,
    // Controla que se procesen los 8 datos de la ventana
    Load contador
    Subt ONE
    SkipCond 400
    Jump Sumar
    Jump Dividir

Sumar,
    // Avanza al siguiente dato
    Load ptr_Datos
    Add ONE
    Store ptr_Datos

    // Agrega el dato actual a la suma
    Load suma
    AddI ptr_Datos
    Store suma

    // Disminuye el contador de datos pendientes
    Load contador
    Subt ONE
    Store contador
    Jump Contador

Dividir,
    // Calcula el promedio mediante restas sucesivas
    // Cada resta del divisor (8) representa una unidad
    // del resultado final.
    Load suma
    Subt divisor
    SkipCond 000
    Jump Restar
    Jump Suavizar

Restar,
    // Cuenta cuántas veces se puede restar 8 de la suma
    Load resta
    Add ONE
    Store resta

    Load suma
    Subt divisor
    Store suma
    Jump Dividir

Suavizar,
    // Muestra el promedio calculado
    Load resta
    Output

    // Guarda el promedio en la posición correspondiente
    // del arreglo Moving.
    Load ptr_Moving
    Add inicio_movil
    Store ptr_Moving

    LoadI ptr_Moving
    Add resta
    StoreI ptr_Moving

    // Avanza la ventana móvil una posición
    Load inicio_movil
    Add ONE
    Store inicio_movil

    // Restaura el contador a 8 para la siguiente ventana
    Load contador
    Subt ONE
    Add divisor
    Store contador

    // Reduce la cantidad de datos pendientes
    Load num_datos
    Subt ONE
    Store num_datos

    // Procesa la siguiente ventana
    Jump Inicio

Fin,
    Halt



// VARIABLES Y CONSTANTES

num_suavizado, DEC 0    // Cantidad de valores suavizados
num_datos, DEC 193      // Cantidad de datos a procesar
inicio_movil, DEC 0     // Desplazamiento de la ventana móvil
mover, DEC 0            // Variable auxiliar
suma, DEC 0             // Suma de los datos de la ventana
resta, DEC 0            // Resultado del promedio por restas sucesivas
ONE, DEC 1              // Constante para incrementar/decrementar
divisor, DEC 8          // Tamaño de la ventana
contador, DEC 8         // Datos que faltan por sumar en la ventana

// Punteros a las diferentes zonas de memoria
ptr_Datos, adr Datos
ptr_Moving, adr Moving
ptr_inicial, adr Inicial
