Inicio,Load num_datos
	Subt ONE
	//Output
	SkipCond 400
	Jump Inicializar
	Jump Fin
Inicializar, Load ptr_inicial
	Add inicio_movil
	Store ptr_Datos
	Load suma
	Clear
	//Output
	Store suma
	Load resta
	Clear
	//Output
	Store resta
	LoadI ptr_Datos
	//Output
	Store suma
	Jump Contador
Contador, Load contador
	Subt ONE
	SkipCond 400
	Jump Sumar
	Jump Dividir
Sumar, Load ptr_Datos
	Add ONE
	Store ptr_Datos
	Load suma
	AddI ptr_Datos
	Store suma
	//Output
	Load contador
	Subt ONE
	Store contador
	Jump Contador

Dividir, Load suma
	//Output
	Subt divisor
	SkipCond 000
	Jump Restar
	Jump Suavizar
Restar, Load resta
	Add ONE
	Store resta
	Load suma
	Subt divisor
	Store suma
	Jump Dividir
	
Suavizar,Load resta
	Output
	Load ptr_Moving
	Add inicio_movil
	Store ptr_Moving
	LoadI ptr_Moving
	Add resta
	StoreI ptr_Moving
	//Output
	Load inicio_movil
	Add ONE
	Store inicio_movil
	Load contador 
	Subt ONE
	Add divisor
	Store contador
	Load num_datos
	Subt ONE
	Store num_datos
	Jump Inicio
Fin, Halt

resta, DEC 0
ONE, DEC 1
divisor, DEC 8
contador, DEC 8

ptr_Moving, adr Moving
